! ===== Switch5 : VTP CLIENT, ROOT BRIDGE VLAN 40 =====
enable
configure terminal
hostname Switch5
no ip domain-lookup

! ---- VTP ----
vtp domain cisco.com
vtp mode client
vtp version 2
vtp password cisco123

! ---- Trunks: Fa0/1->MLS  Fa0/2->Switch6  Fa0/3->Switch4 ----
interface range fastEthernet 0/1 - 3
 switchport mode trunk
 no shutdown
exit

! ---- STP: root for VLAN 40, backup root for VLAN 30 ----
spanning-tree mode pvst
spanning-tree vlan 40 priority 4096
spanning-tree vlan 30 priority 8192
end
write memory
