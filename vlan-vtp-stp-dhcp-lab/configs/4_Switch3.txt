! ===== Switch3 : VTP TRANSPARENT, ROOT BRIDGE VLAN 20 =====
enable
configure terminal
hostname Switch3
no ip domain-lookup

! ---- VTP ----
vtp domain cisco.com
vtp mode transparent
vtp version 2
vtp password cisco123

! ---- VLANs created LOCALLY (transparent mode never syncs) ----
vlan 10
 name marketing
vlan 20
 name sales
vlan 30
 name engineering
vlan 40
 name finance
exit

! ---- Trunks: Fa0/1->Switch1  Fa0/2->MLS  Fa0/3->Switch2 ----
interface range fastEthernet 0/1 - 3
 switchport mode trunk
 no shutdown
exit

! ---- STP: root for VLAN 20, backup root for VLAN 10 ----
spanning-tree mode pvst
spanning-tree vlan 20 priority 4096
spanning-tree vlan 10 priority 8192
end
write memory
