! ===== Switch4 : VTP TRANSPARENT, ROOT BRIDGE VLAN 30 =====
enable
configure terminal
hostname Switch4
no ip domain-lookup

! ---- VTP ----
vtp domain cisco.com
vtp mode transparent
vtp version 2
vtp password cisco123

! ---- VLANs created LOCALLY (transparent mode never syncs) ----
vlan 10
 name marketing
vlan 20
 name sales
vlan 30
 name engineering
vlan 40
 name finance
exit

! ---- Trunks: Fa0/1->MLS  Fa0/2->Switch6 (Fa0/1)  Fa0/3->Switch5 ----
interface range fastEthernet 0/1 - 3
 switchport mode trunk
 no shutdown
exit

! ---- STP: root for VLAN 30, backup root for VLAN 40 ----
spanning-tree mode pvst
spanning-tree vlan 30 priority 4096
spanning-tree vlan 40 priority 8192
end
write memory
