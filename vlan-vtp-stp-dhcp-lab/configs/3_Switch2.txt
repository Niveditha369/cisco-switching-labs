! ===== Switch2 : VTP CLIENT, ROOT BRIDGE VLAN 10 =====
enable
configure terminal
hostname Switch2
no ip domain-lookup

! ---- VTP ----
vtp domain cisco.com
vtp mode client
vtp version 2
vtp password cisco123

! ---- Trunks: Fa0/1->Switch1  Fa0/2->MLS  Fa0/3->Switch3 ----
interface range fastEthernet 0/1 - 3
 switchport mode trunk
 no shutdown
exit

! ---- STP: root for VLAN 10, backup root for VLAN 20 ----
spanning-tree mode pvst
spanning-tree vlan 10 priority 4096
spanning-tree vlan 20 priority 8192
end
write memory
